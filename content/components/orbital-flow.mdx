## Usage

<span className="text-[15px]">
  Follow the steps below to add the Orbital Flow Component to your project.
</span>

{/* -----------------------------USAGE - STEP - 01 ---------------------------> */}

<UsageSteps stepNumber={1} title="Add the import statement for the component">
  ```tsx 
  import OrbitalFlow from "@/components/elixir-ui/orbital-flow"; 
  ```
</UsageSteps>

{/* -----------------------------USAGE - STEP - 02 ---------------------------> */}

<UsageSteps stepNumber={2} title="Add the inner and outer tags props to the component">
  ```tsx
  const innerTags = [
    { name: "Slack", icon: <FaSlack /> },
    { name: "Notion", icon: <SiNotion /> },
    { name: "Discord", icon: <FaDiscord /> },
    { name: "Youtube", icon: <FaYoutube /> },
  ];

const outerTags = [
{ name: "Messages", icon: <FaEnvelope /> },
{ name: "Substack", icon: <FaRss /> },
{ name: "Recordings", icon: <FaMicrophone /> },
{ name: "Spotify", icon: <FaSpotify /> },
{ name: "Instagram", icon: <FaInstagram /> },
];

````
</UsageSteps>

{/* -----------------------------USAGE - STEP - 03 ---------------------------> */}
<UsageSteps stepNumber={2} title="Import the component in your project and use it">
```tsx
<div className="h-full w-full flex items-center justify-center">
    <OrbitalFlow
          outerTags={outerTags}
          innerTags={innerTags}
          centerImageSrc="/components/john-lennon.jpg"
          outerOrbitDiameter={850}
          innerOrbitDiameter={550}
          outerRotationSpeed={32}
          innerRotationSpeed={30}
    />
</div>
````

</UsageSteps>
