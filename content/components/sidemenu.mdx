  Follow the steps below to add the Sidemenu Component to your project.

{/* -----------------------------USAGE - STEP - 01 ---------------------------> */}

<UsageSteps stepNumber={1} title="Add the import statement for the component">

  ```tsx 
  import { Sidemenu } from "@/components/elixir-ui/sidemenu"; 
  ```
</UsageSteps>

{/* -----------------------------USAGE - STEP - 02 ---------------------------> */}

<UsageSteps stepNumber={2} title="Add the links and footer links props to the component">

  ```tsx
    { title: "Projects", href: "#" },
    { title: "Agency", href: "#" },

{ title: "Facebook", href: "#" },
]

````
</UsageSteps>

{/* -----------------------------USAGE - STEP - 03 ---------------------------> */}
<UsageSteps stepNumber={3} title="Import the component in your project and use it">
```tsx
<div className="absolute top-10 right-10 flex justify-end h-full w-full">
  <Sidemenu
    imageSaturation={1.2}

    footerLinks={footerLinks}
  />
</div>
````

</UsageSteps>
