<UsageSteps
  stepNumber={3}
  title="Import the component in your project and use it"

  ```tsx
  "use client"

  import { ActionButton, ActionButton2 } from "@/components/elixir-ui/action-button";
  import { useState } from "react";

  export default function Home() {
    const [isActive, setIsActive] = useState(false);
    return (
      <div className="flex items-center justify-center min-h-screen">
        <ActionButton isActive={isActive} setIsActive={setIsActive} />
        <ActionButton2 className="rounded-full">Get Started</ActionButton2>
      </div>
    )
  }
  ```
</UsageSteps>
