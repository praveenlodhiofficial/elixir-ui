## Usage

### Basic usage

```tsx
import { MotionSidebar } from "@/components/ui/motion-sidebar";

export default function DemoPage() {
  return <MotionSidebar />;
}
```

### Custom links

```tsx
import { MotionSidebar } from "@/components/ui/motion-sidebar";

const links = [
  { title: "Home", href: "/", type: "main" },
  { title: "Work", href: "/work", type: "main" },
  { title: "About", href: "/about", type: "main" },
  { title: "X", href: "https://x.com", type: "footer" },
  { title: "GitHub", href: "https://github.com", type: "footer" },
];

export default function DemoPage() {
  return <MotionSidebar links={links} className="z-50" />;
}
```
