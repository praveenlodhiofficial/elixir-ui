## Usage

```tsx
import { GalleryShowcase } from "@/components/ui/gallery-showcase";

export default function DemoPage() {
  return (
    <GalleryShowcase
      images={[
        "/path/to/image-1.jpg",
        "/path/to/image-2.jpg",
        "/path/to/image-3.jpg",
      ]}
    />
  );
}
```
