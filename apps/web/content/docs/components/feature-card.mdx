import { FeatureCard } from "@workspace/ui/components/feature-card";

### Available Variants

- `background` – Image-based highlight card
- `stat` – Numeric/statistics card
- `media` – Image + content card
- `text` – Quote/content card

Use the `variant` prop to switch between styles.

## Usage

### Background Card

```tsx
import { FeatureCard } from "@workspace/ui/components/feature-card";

export function BackgroundCardDemo() {
  return (
    <FeatureCard
      variant="background"
      bgImage="https://images.unsplash.com/photo-1500534314209-a25ddb2bd429?q=80&w=1200&auto=format&fit=crop"
      subtitle="Case Study"
      description="Design is intelligence made visible"
      title="Elixir UI"
      statLabel="2026"
    />
  );
}
```

### Stat Card

```tsx
import { FeatureCard } from "@workspace/ui/components/feature-card";

export function StatCardDemo() {
  return (
    <FeatureCard
      variant="stat"
      subtitle="Users"
      statValue="10K+"
      description="Active monthly users"
    />
  );
}
```

### Media Card

```tsx
import { FeatureCard } from "@workspace/ui/components/feature-card";

export function MediaCardDemo() {
```
