## Usage

```tsx
import { StatCard } from "@workspace/ui/components/stat-card";

export function Demo() {
  return (
    <StatCard
      variant="colorful"
      label="Conversions"
      value="3.8×"
      caption="vs industry avg"
      badge="↑ 22%"
      badgeTone="positive"
    />
  );
}
```
